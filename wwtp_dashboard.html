/** * ================================================================ * WWTP DASHBOARD - BACKEND (Google Apps Script) - v2 * Trạm xử lý nước thải YOU YOUNG VINA * ================================================================ * CÁCH CÀI ĐẶT: giống hướng dẫn trước — dán toàn bộ file này vào * Apps Script (Tiện ích mở rộng > Apps Script), Lưu, rồi Triển khai * lại (Deploy > Manage deployments > Edit > chọn "New version" > Deploy) * để bản cập nhật có hiệu lực trên URL cũ. * ================================================================ */ var SHEET_USERS = 'Users'; var SHEET_DAILY_ENTRIES = 'DailyEntries'; function doGet(e) { var action = e.parameter.action; if (action === 'getData') { return jsonResponse(getAllData()); } return jsonResponse({ error: 'Invalid or missing action parameter' }); } function doPost(e) { var body; try { body = JSON.parse(e.postData.contents); } catch (err) { return jsonResponse({ success: false, message: 'Dữ liệu gửi lên không hợp lệ' }); } if (body.action === 'login') { return jsonResponse(checkLogin(body.username, body.password)); } if (body.action === 'saveEntry') { return jsonResponse(saveDailyEntry(body.entry)); } return jsonResponse({ success: false, message: 'Invalid action' }); } function checkLogin(username, password) { if (!username || !password) { return { success: false, message: 'Vui lòng nhập tài khoản và mật khẩu' }; } var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_USERS); if (!sheet) return { success: false, message: 'Chưa tạo tab Users' }; var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { var row = data[i]; if (String(row[0]).trim() === String(username).trim() && String(row[1]).trim() === String(password).trim()) { return { success: true, username: row[0], fullname: row[2], role: row[3] }; } } return { success: false, message: 'Sai tài khoản hoặc mật khẩu' }; } /** * Ghi 1 dòng báo cáo ca vào tab DailyEntries. * entry là 1 object JSON gửi từ dashboard, các field tương ứng đúng thứ tự cột. * Nếu tab DailyEntries chưa có, hàm sẽ tự tạo tab + tiêu đề cột. */ function saveDailyEntry(entry) { if (!entry) return { success: false, message: 'Không có dữ liệu để lưu' }; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_DAILY_ENTRIES); var headers = [ 'timestamp','date','shift','username','fullname', 'influent_m3','ro_permeate_m3','pump_to_yy_m3','recovery_pct', 'ph_ro','tds_ro','orp_ro','color_ro','turbidity_ro', 'chem_pac','chem_polymer','chem_naoh','chem_javen','chem_hcl','chem_citric','chem_genesys', 'tasks_done','notes' ]; if (!sheet) { sheet = ss.insertSheet(SHEET_DAILY_ENTRIES); sheet.appendRow(headers); } var row = headers.map(function(h){ if (h === 'timestamp') return new Date().toISOString(); return entry[h] !== undefined ? entry[h] : ''; }); sheet.appendRow(row); return { success: true, message: 'Đã lưu báo cáo ca thành công' }; } function sheetToObjects(sheetName) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName); if (!sheet) return []; var values = sheet.getDataRange().getValues(); if (values.length < 2) return []; var headers = values[0]; var rows = []; for (var i = 1; i < values.length; i++) { var obj = {}; var isEmpty = true; for (var j = 0; j < headers.length; j++) { obj[headers[j]] = values[i][j]; if (values[i][j] !== '' && values[i][j] !== null) isEmpty = false; } if (!isEmpty) rows.push(obj); } return rows; } // Lấy N dòng cuối cùng của 1 tab (dùng cho DailyEntries - chỉ cần xem gần đây) function sheetToObjectsLastN(sheetName, n) { var all = sheetToObjects(sheetName); if (all.length <= n) return all; return all.slice(all.length - n); } function getAllData() { return { kpis: sheetToObjects('KPIs'), monthlyTrend: sheetToObjects('MonthlyTrend'), influent: sheetToObjects('Influent'), operatingSummary: sheetToObjects('OperatingSummary'), lineContribution: sheetToObjects('LineContribution'), costVariance: sheetToObjects('CostVariance'), compliance: sheetToObjects('Compliance'), chemicalUsage: sheetToObjects('ChemicalUsage'), chemicalInventory: sheetToObjects('ChemicalInventory'), chemicalStockIn: sheetToObjects('ChemicalStockIn'), dailyEntries: sheetToObjectsLastN('DailyEntries', 30), updatedAt: new Date().toISOString() }; } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
